        function playSound(type) {
            if(!soundEnabled || audioCtx.state === 'suspended') return;
            const osc = audioCtx.createOscillator();
            const gainNode = audioCtx.createGain();
            osc.connect(gainNode);
            gainNode.connect(audioCtx.destination);
            
            if(type === 'correct') {
                osc.type = 'sine'; osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1046.50, audioCtx.currentTime + 0.3);
                gainNode.gain.setValueAtTime(0.5, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                osc.start(); osc.stop(audioCtx.currentTime + 0.3);
            } else if (type === 'wrong') {
                osc.type = 'sawtooth'; osc.frequency.setValueAtTime(150, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.3);
                gainNode.gain.setValueAtTime(0.5, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                osc.start(); osc.stop(audioCtx.currentTime + 0.3);
            } else if (type === 'slice') {
                osc.type = 'square'; osc.frequency.setValueAtTime(800, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.1);
                gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
                osc.start(); osc.stop(audioCtx.currentTime + 0.1);
            }
        }

            const level = levelData[currentLevelIndex];
            
            // 排序庫存，讓大晶塊排前面
            zone2_inventory.sort((a,b)=>b-a);

            let invHTML = zone2_inventory.map((val, idx) => {
                let cls = val === 60 ? 'crystal-hr' : (val === 10 ? 'crystal-min10' : 'crystal-min1');
                let label = val === 60 ? '1 小時' : (val === 10 ? '10 分' : '1 分');
                return `<div class="crystal-block ${cls}" draggable="true" ondragstart="handleDragStart(event, 'inv', ${idx})" onclick="handleZone2Click('inv', ${idx})">${label}</div>`;
            }).join('');

            let fragmentsHTML = zone2_placed.map((fragArray, fIdx) => {
                fragArray.sort((a,b)=>b-a);
                let placedHTML = fragArray.map((val, itemIdx) => {
                    let cls = val === 60 ? 'crystal-hr' : (val === 10 ? 'crystal-min10' : 'crystal-min1');
                    let label = val === 60 ? '1 小時' : (val === 10 ? '10 分' : '1 分');
                    return `<div class="crystal-block ${cls}" draggable="true" ondragstart="handleDragStart(event, 'frag_${fIdx}', ${itemIdx})" onclick="handleZone2Click('frag_${fIdx}', ${itemIdx})">${label}</div>`;
                }).join('');
                
                if (fragArray.length === 0) placedHTML = '<span style="color:gray;">拖曳至此</span>';
                
                return `<div class="target-fragment" id="z2-target-${fIdx}" ondragover="allowDrop(event)" ondrop="handleDrop(event, 'frag_${fIdx}')">${placedHTML}</div>`;
            }).join('');

            inputArea.innerHTML = `
                <div style="display:flex; justify-content:center; align-items:center; width:100%; margin-bottom:5px; gap: 15px; flex-wrap: wrap;">
                    <button class="fuse-btn" onclick="fuseBlocks(10)">🔄 合併 10 個 [1 分] 為 [10 分]</button>
                    <div style="color:var(--brand-orange); font-size:16px; font-weight:bold;">能量儲存區 (點擊晶塊可向下切割)</div>
                    <button class="fuse-btn" onclick="fuseBlocks(60)">🔄 合併 6 個 [10 分] 為 [1 小時]</button>
                </div>
                <div class="crystal-field" id="z2-inv" ondragover="allowDrop(event)" ondrop="handleDrop(event, 'inv')">${invHTML}</div>
                <div style="color:var(--neon-blue); margin-top:10px;">待分配碎片區</div>
                <div class="fragment-container">${fragmentsHTML}</div>
            `;
        }

        function fuseBlocks(targetVal) {
            let needed = targetVal === 60 ? 6 : 10;
            let searchVal = targetVal === 60 ? 10 : 1;
            
            let foundCount = zone2_inventory.filter(v => v === searchVal).length;
            if(foundCount >= needed) {
                playSound('slice');
                for(let i=0; i<needed; i++) {
                    zone2_inventory.splice(zone2_inventory.indexOf(searchVal), 1);
                }
                zone2_inventory.push(targetVal);
                showMessage(`🔄 晶體融合成功！`);
                renderZone2UI();
            } else {
                showMessage(`⚠️ 儲存區數量不足！需要 ${needed} 個 [${searchVal}分]`);
            }
        }

        function handleDragStart(e, source, index) {
            e.dataTransfer.setData('source', source);
            e.dataTransfer.setData('index', index);
        }
        function allowDrop(e) { e.preventDefault(); }
        function handleDrop(e, target) {
            e.preventDefault();
            let source = e.dataTransfer.getData('source');
            let index = parseInt(e.dataTransfer.getData('index'));
            if(source === target || isNaN(index)) return;

            let val;
            if (source === 'inv') {
                val = zone2_inventory.splice(index, 1)[0];
            } else if (source.startsWith('frag_')) {
                let fIdx = parseInt(source.split('_')[1]);
                val = zone2_placed[fIdx].splice(index, 1)[0];
            }

            if (target === 'inv') {
                zone2_inventory.push(val);
            } else if (target.startsWith('frag_')) {
                let fIdx = parseInt(target.split('_')[1]);
                zone2_placed[fIdx].push(val);
            }
            playSound('slice'); 
            renderZone2UI();
        }

        function handleZone2Click(source, index) {
            let targetArray = source === 'inv' ? zone2_inventory : zone2_placed[parseInt(source.split('_')[1])];
            let val = targetArray[index];
            
            if(val === 60) {
                playSound('slice');
                targetArray.splice(index, 1);
                for(let i=0; i<6; i++) targetArray.push(10);
                showMessage("⚡ 晶體切割：1小時 轉為 6個10分！");
                renderZone2UI();
            } else if (val === 10) {
                playSound('slice');
                targetArray.splice(index, 1);
                for(let i=0; i<10; i++) targetArray.push(1);
                showMessage("⚡ 晶體切割：10分 轉為 10個1分！");
                renderZone2UI();
            }
        }

        function showMessage(msg, duration = 2500) {
            const el = document.getElementById('combo-display');
            el.innerText = msg;
            setTimeout(() => { if(el.innerText === msg) el.innerText = ''; }, duration);
        }

        // --- 答案驗證與自動進入下一題 ---
        function checkAnswer() {
            if(isPaused) return;
            const level = levelData[currentLevelIndex];
            let isCorrect = false;

            document.getElementById('whiteboard-container').style.display = 'none';

            if(level.type === 'zone1') {
                let r1 = parseInt(document.getElementById('z1-raw-1').value);
                let r2 = parseInt(document.getElementById('z1-raw-2').value);
                let a1 = parseInt(document.getElementById('slide-1').value);
                let a2 = parseInt(document.getElementById('slide-2').value);
                
                if(r1 === level.raw1 && r2 === level.raw2 && a1 === level.ans1 && a2 === level.ans2) {
                    isCorrect = true;
                }
            } 
            else if(level.type === 'zone2') {
                let allCorrect = true;
                for(let i=0; i<zone2_placed.length; i++) {
                    let total = zone2_placed[i].reduce((a, b) => a + b, 0);
                    if (total !== level.targetTotalMins) {
                        allCorrect = false;
                        break;
                    }
                }
                if(allCorrect && zone2_placed.length > 0) isCorrect = true;
            }
            else if(level.type === 'zone3') {
                if(level.subType === 'A') {
                    let tm = parseInt(document.getElementById('z3a-t-m').value) || 0;
                    let ts = parseInt(document.getElementById('z3a-t-s').value) || 0;
                    let sm = parseInt(document.getElementById('z3a-s-m').value) || 0;
                    let ss = parseInt(document.getElementById('z3a-s-s').value) || 0;
                    let ansT = parseInt(document.getElementById('z3a-ans-times').value);
                    
                    let convT = parseInt(document.getElementById('z3a-conv-t').value);
                    let convS = parseInt(document.getElementById('z3a-conv-s').value);
                    let ft = parseInt(document.getElementById('z3a-final-t').value);
                    let fs = parseInt(document.getElementById('z3a-final-s').value);
                    let fans = parseInt(document.getElementById('z3a-final-ans').value);

                    let correctTotalS = level.totalM_disp * 60 + level.totalS_disp;
                    let correctSingleS = level.singleM * 60 + level.singleS;

                    if(tm === level.totalM_disp && ts === level.totalS_disp && 
                       sm === level.singleM && ss === level.singleS && ansT === level.ansTimes &&
                       convT === correctTotalS && convS === correctSingleS &&
                       ft === correctTotalS && fs === correctSingleS && fans === level.ansTimes) {
                        isCorrect = true;
                    }
                } else {
                    let g = parseInt(document.getElementById('z3b-gaps').value);
                    let tm = parseInt(document.getElementById('z3b-total-m').value);
                    let ap = document.getElementById('z3b-ans-period').value;
                    let ah = parseInt(document.getElementById('z3b-ans-h').value);
                    let am = parseInt(document.getElementById('z3b-ans-m').value);
                    
                    if(g === level.ansGaps && tm === level.ansTotalM && 
                       ap === level.ansPeriod && ah === level.ansH && am === level.ansM) {
                        isCorrect = true;
                    }
                }
            }

            if(isCorrect) {
                score++;
                updateHUD();
                document.getElementById('main-submit-btn').style.display = 'none';

                if(level.type === 'zone1') {
                    playSound('correct');
                    document.getElementById('z1-orbit').classList.add('beam-active');
                    showMessage("✨ 光譜連線成功！軌道校準完成！");
                    setTimeout(nextLevel, 2000);
                } 
                else if(level.type === 'zone2') {
                    playSound('correct');
                    zone2_placed.forEach((frag, idx) => {
                        document.getElementById(`z2-target-${idx}`).classList.add('completed');
                    });
                    showMessage("🔗 能量分配完美！星環修復！");
                    setTimeout(nextLevel, 2000);
                } 
                else if(level.type === 'zone3') {
                    let bh = document.getElementById('z3-visual-bh');
                    bh.style.display = 'block';
                    playSound('correct');
                    
                    let threw = 0;
                    let throwInt = setInterval(() => {
                        if(threw >= 15) {
                            clearInterval(throwInt);
                            document.getElementById('game-screen').classList.add('space-beam');
                            blackHoleSafeZone = 150;
                            updateBlackHoleCSS();
                            showMessage("💥 邏輯陣列啟動！黑洞退散！");
                            setTimeout(nextLevel, 2000);
                        } else {
                            let particle = document.createElement('div');
                            particle.className = 'throw-item';
                            particle.style.left = (50 + (Math.random()*40 - 20)) + 'px';
                            bh.appendChild(particle);
                            setTimeout(()=>particle.remove(), 300);
                            threw++;
                        }
                    }, 80);
                }
            } else {
                playSound('wrong');
                document.getElementById('main-submit-btn').style.display = 'none'; 
                
                let correctMsg = "";
                if(level.type === 'zone1') {
                    correctMsg = `⚠️ 答錯！正確：紅框 ${level.raw1}${level.opt.u1} ${level.raw2}${level.opt.u2}，滑桿 ${level.ans1}${level.opt.u1} ${level.ans2}${level.opt.u2}`;
                } else if(level.type === 'zone2') {
                    const targets = document.querySelectorAll('.target-fragment');
                    targets.forEach(t => {
                        t.classList.add('energy-shock');
                        setTimeout(()=>t.classList.remove('energy-shock'), 500);
                    });
                    correctMsg = `⚠️ 答錯！請確認所有碎片是否精準分配能量`;
                } else if(level.type === 'zone3') {
                    if(level.subType === 'A') correctMsg = `⚠️ 答錯！請確認「所有換算步驟」與最終答案(${level.ansTimes} 次)是否正確。`;
                    else correctMsg = `⚠️ 答錯！正確：間隔 ${level.ansGaps}次，總耗時 ${level.ansTotalM}分，時刻 ${level.ansPeriod} ${level.ansH}時 ${level.ansM}分`;
                }
                
                showMessage(correctMsg, 4000);
                setTimeout(nextLevel, 4000);
            }
        }

        function nextLevel() {
            document.getElementById('game-screen').classList.remove('space-beam');
            currentLevelIndex++;
            if(currentLevelIndex >= levelData.length) {
                endGameSequence();
            } else {
                loadLevel();
            }
        }

        function endGameSequence() {
            clearInterval(timerInterval);
            if(score >= 5) {
                document.getElementById('star-gate').classList.add('gate-opening');
                setTimeout(() => { endGame(); }, 2000);
            } else {
                endGame();
            }
        }

        function updateHUD() {
            document.getElementById('score-display').innerText = `修復進度：${score} / 10`;
        }

        function updateBlackHoleCSS() {
            const bh = document.getElementById('black-hole-layer');
            bh.style.setProperty('--safe-zone', `${blackHoleSafeZone}%`);
            bh.style.setProperty('--danger-zone', `${blackHoleSafeZone + 20}%`);
        }

        function startTimer() {
            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                if(!isPaused) {
                    timeLeft--;
                    let m = Math.floor(timeLeft / 60).toString().padStart(2, '0');
                    let s = (timeLeft % 60).toString().padStart(2, '0');
                    document.getElementById('timer-display').innerText = `時空崩塌倒數：${m}:${s}`;
                    
                    if(currentLevelIndex >= 6) { // 後期加速黑洞
                        blackHoleSafeZone -= 0.5;
                        if(blackHoleSafeZone < 10) blackHoleSafeZone = 10;
                        updateBlackHoleCSS();
                    }

                    if(timeLeft <= 0) {
                        isPaused = true;
                        endGameSequence();
                    }
                }
            }, 1000);
        }

        function togglePause() {
            isPaused = !isPaused;
            document.getElementById('pause-overlay').style.display = isPaused ? 'flex' : 'none';
        }

        // ================= 排行榜系統 =================
        function saveRecord(status) {
            let records = JSON.parse(localStorage.getItem('jessieMathRecords2')) || [];
            const now = new Date();
            const timeStr = `${now.getMonth()+1}/${now.getDate()} ${now.getHours()}:${now.getMinutes().toString().padStart(2,'0')}`;
            records.push({ name: currentPlayer, score: score, status: status, time: timeStr });
            records.sort((a, b) => b.score - a.score);
            localStorage.setItem('jessieMathRecords2', JSON.stringify(records));
        }

        function renderLeaderboard() {
            const tbody = document.getElementById('leaderboard-body');
            tbody.innerHTML = '';
            let records = JSON.parse(localStorage.getItem('jessieMathRecords2')) || [];
            const medals = ["🥇", "🥈", "🥉"];
            
            records.slice(0, 10).forEach((rec, index) => {
                const rankIcon = index < 3 ? medals[index] : `🎖️ ${index+1}`;
                tbody.innerHTML += `
                    <tr>
                        <td>${rankIcon}</td><td>${rec.name}</td><td>${rec.score} / 10</td>
                        <td style="color:${rec.status==='任務成功'?'#2ecc71':'#e74c3c'}">${rec.status}</td>
                        <td>${rec.time}</td>
                    </tr>
                `;
            });
        }

        // ================= 白板功能 (Canvas) =================
        const canvas = document.getElementById('draw-canvas');
        const ctx = canvas.getContext('2d');
        let isDrawing = false;
        let currentDrawMode = 'pen';

        function setDrawMode(mode) {
            currentDrawMode = mode;
            
            document.querySelectorAll('.wb-controls button').forEach(b => b.classList.remove('wb-active-btn'));
            document.getElementById('wb-btn-' + mode).classList.add('wb-active-btn');

            if (mode === 'mouse') {
                canvas.style.pointerEvents = 'none';
            } else {
                canvas.style.pointerEvents = 'auto';
                if (mode === 'pen') {
                    ctx.globalCompositeOperation = 'source-over';
                    ctx.strokeStyle = "#ffffff";
                    ctx.lineWidth = 3; 
                } else if (mode === 'eraser') {
                    ctx.globalCompositeOperation = 'destination-out';
                    ctx.lineWidth = 30; 
                }
            }
        }

        function resizeCanvas() {
            const container = document.getElementById('whiteboard-container');
            if(!container) return;
            canvas.width = container.clientWidth;
            canvas.height = container.clientHeight;
            ctx.lineCap = "round";
            ctx.lineJoin = "round";
            setDrawMode(currentDrawMode);
        }

        function toggleWhiteboard() {
            const wb = document.getElementById('whiteboard-container');
            if(wb.style.display === 'block') { 
                wb.style.display = 'none'; 
            } else { 
                wb.style.display = 'block'; 
                resizeCanvas(); 
                setDrawMode('pen'); 
            }
        }

        function clearWhiteboard() { 
            ctx.clearRect(0, 0, canvas.width, canvas.height); 
        }

        const startPos = (e) => { 
            if(currentDrawMode === 'mouse') return;
            isDrawing = true; 
            draw(e); 
        };
        const endPos = () => { 
            isDrawing = false; 
            ctx.beginPath(); 
        };
        const draw = (e) => {
            if(!isDrawing || currentDrawMode === 'mouse') return; 
            if(e.target === canvas) e.preventDefault();
            
            const rect = canvas.getBoundingClientRect();
            const clientX = e.clientX || (e.touches && e.touches[0] ? e.touches[0].clientX : 0);
            const clientY = e.clientY || (e.touches && e.touches[0] ? e.touches[0].clientY : 0);
            const x = clientX - rect.left;
            const y = clientY - rect.top;
            
            ctx.lineTo(x, y); ctx.stroke(); ctx.beginPath(); ctx.moveTo(x, y);
        };

        canvas.addEventListener('mousedown', startPos); 
        canvas.addEventListener('mouseup', endPos);
        canvas.addEventListener('mousemove', draw); 
        canvas.addEventListener('touchstart', startPos, {passive: false});
        canvas.addEventListener('touchend', endPos); 
        canvas.addEventListener('touchmove', draw, {passive: false});

        window.addEventListener('resize', () => {
            if (document.getElementById('whiteboard-container').style.display === 'block') {
                const tempCanvas = document.createElement('canvas');
                const tempCtx = tempCanvas.getContext('2d');
                tempCanvas.width = canvas.width;
                tempCanvas.height = canvas.height;
                tempCtx.drawImage(canvas, 0, 0);
                
                resizeCanvas();
                ctx.drawImage(tempCanvas, 0, 0);
            }
        });

        window.onload = () => { renderLeaderboard(); };
    </script>
